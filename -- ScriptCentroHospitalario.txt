-- Script

CREATE DATABASE CentroHospitalario;

-- 1. MÓDULO DE CENTROS MÉDICOS

CREATE TABLE CentroMedico (
    CentroID INT PRIMARY KEY IDENTITY(1,1),
    Nombre VARCHAR(100) NOT NULL,
    Ciudad VARCHAR(100) NOT NULL,
    Direccion VARCHAR(150) NOT NULL,
    Telefono VARCHAR(20) NOT NULL,
    Email VARCHAR(100) NOT NULL,
    Sede VARCHAR(50) NOT NULL,
    FechaCreacion DATETIME DEFAULT GETDATE(),
    Estado BIT DEFAULT 1, -- 1: Activo, 0: Inactivo
    CONSTRAINT UQ_Centro_Email UNIQUE(Email),
    CONSTRAINT CK_Centro_Sede CHECK (Sede IN ('HUACHI', 'INGAHURCO', 'QUEROCHACA'))
);

-- 2. MÓDULO DE ESPECIALIDADES

CREATE TABLE Especialidad (
    EspecialidadID INT PRIMARY KEY IDENTITY(1,1),
    Nombre VARCHAR(100) NOT NULL UNIQUE,
    Descripcion VARCHAR(255),
    Estado BIT DEFAULT 1
);

-- 3. MÓDULO DE MÉDICOS

CREATE TABLE Medico (
    MedicoID INT PRIMARY KEY IDENTITY(1,1),
    Cedula VARCHAR(10) NOT NULL UNIQUE,
    Nombres VARCHAR(100) NOT NULL,
    Apellidos VARCHAR(100) NOT NULL,
    Email VARCHAR(100) NOT NULL UNIQUE,
    Telefono VARCHAR(20) NOT NULL,
    FechaIngreso DATE NOT NULL DEFAULT CAST(GETDATE() AS DATE),
    CentroID INT NOT NULL,
    Estado BIT DEFAULT 1,
    FechaCreacion DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Medico_Centro FOREIGN KEY (CentroID) 
        REFERENCES CentroMedico(CentroID) ON DELETE CASCADE,
    CONSTRAINT CK_Medico_Cedula CHECK (LEN(Cedula) = 10)
);

 Indices para optimización

CREATE INDEX IX_Medico_Centro ON Medico(CentroID);
CREATE INDEX IX_Medico_Cedula ON Medico(Cedula);

Tabla intermedia: MedicoEspecialidad (N:M)

CREATE TABLE MedicoEspecialidad (
    MedicoID INT NOT NULL,
    EspecialidadID INT NOT NULL,
    FechaAsignacion DATE DEFAULT CAST(GETDATE() AS DATE),
    Principal BIT DEFAULT 0, --> Indica si es la especialidad principal
    CONSTRAINT PK_MedicoEspecialidad PRIMARY KEY (MedicoID, EspecialidadID),
    CONSTRAINT FK_MedicoEsp_Medico FOREIGN KEY (MedicoID) 
        REFERENCES Medico(MedicoID) ON DELETE CASCADE,
    CONSTRAINT FK_MedicoEsp_Especialidad FOREIGN KEY (EspecialidadID) 
        REFERENCES Especialidad(EspecialidadID) ON DELETE CASCADE
);

-- 4. MÓDULO DE EMPLEADOS

Tabla: Empleado
CREATE TABLE Empleado (
    EmpleadoID INT PRIMARY KEY IDENTITY(1,1),
    Cedula VARCHAR(10) NOT NULL UNIQUE,
    Nombres VARCHAR(100) NOT NULL,
    Apellidos VARCHAR(100) NOT NULL,
    Cargo VARCHAR(100) NOT NULL,
    Telefono VARCHAR(20) NOT NULL,
    Email VARCHAR(100) NOT NULL UNIQUE,
    FechaIngreso DATE NOT NULL DEFAULT CAST(GETDATE() AS DATE),
    CentroID INT NOT NULL,
    Estado BIT DEFAULT 1,
    FechaCreacion DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Empleado_Centro FOREIGN KEY (CentroID) 
        REFERENCES CentroMedico(CentroID) ON DELETE CASCADE,
    CONSTRAINT CK_Empleado_Cedula CHECK (LEN(Cedula) = 10)
);

 Índices para optimización
CREATE INDEX IX_Empleado_Centro ON Empleado(CentroID);

-- 5. MÓDULO DE PACIENTES

 Tabla: Paciente
CREATE TABLE Paciente (
    PacienteID INT PRIMARY KEY IDENTITY(1,1),
    Cedula VARCHAR(10) NOT NULL UNIQUE,
    Nombres VARCHAR(100) NOT NULL,
    Apellidos VARCHAR(100) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    Direccion VARCHAR(150) NOT NULL,
    Telefono VARCHAR(20) NOT NULL,
    Email VARCHAR(100),
    Sexo CHAR(1), --> M: Masculino, F: Femenino
    TipoSangre VARCHAR(5), --> A+, O-, etc.
    Estado BIT DEFAULT 1,
    FechaRegistro DATETIME DEFAULT GETDATE(),
    CONSTRAINT CK_Paciente_Cedula CHECK (LEN(Cedula) = 10),
    CONSTRAINT CK_Paciente_Sexo CHECK (Sexo IN ('M', 'F')),
    CONSTRAINT CK_Paciente_FechaNac CHECK (FechaNacimiento < GETDATE())
);

 Índices para optimización
CREATE INDEX IX_Paciente_Cedula ON Paciente(Cedula);

-- 6. MÓDULO DE CONSULTAS MÉDICAS

 Tabla: Consulta
CREATE TABLE Consulta (
    ConsultaID INT PRIMARY KEY IDENTITY(1,1),
    FechaConsulta DATETIME NOT NULL DEFAULT GETDATE(),
    Motivo VARCHAR(255) NOT NULL,
    Diagnostico TEXT,
    Observaciones TEXT,
    MedicoID INT NOT NULL,
    PacienteID INT NOT NULL,
    CentroID INT NOT NULL,
    EstadoConsulta VARCHAR(20) DEFAULT 'PROGRAMADA', --> PROGRAMADA, COMPLETADA, CANCELADA
    FechaCreacion DATETIME DEFAULT GETDATE(),
    CONSTRAINT FK_Consulta_Medico FOREIGN KEY (MedicoID) 
        REFERENCES Medico(MedicoID),
    CONSTRAINT FK_Consulta_Paciente FOREIGN KEY (PacienteID) 
        REFERENCES Paciente(PacienteID),
    CONSTRAINT FK_Consulta_Centro FOREIGN KEY (CentroID) 
        REFERENCES CentroMedico(CentroID),
    CONSTRAINT CK_Consulta_Estado CHECK (EstadoConsulta IN ('PROGRAMADA', 'COMPLETADA', 'CANCELADA'))
);

 Índices para optimización
CREATE INDEX IX_Consulta_Fecha ON Consulta(FechaConsulta);
CREATE INDEX IX_Consulta_Medico ON Consulta(MedicoID);
CREATE INDEX IX_Consulta_Paciente ON Consulta(PacienteID);

-- 7. MÓDULO DE MEDICAMENTOS Y RECETAS

 Tabla: Medicamento
CREATE TABLE Medicamento (
    MedicamentoID INT PRIMARY KEY IDENTITY(1,1),
    Nombre VARCHAR(100) NOT NULL,
    Descripcion VARCHAR(255),
    Fabricante VARCHAR(100),
    Tipo VARCHAR(50), -- Analgésico, Antibiótico, etc.
    PrecioUnitario DECIMAL(10,2),
    Stock INT DEFAULT 0,
    Estado BIT DEFAULT 1,
    CONSTRAINT CK_Medicamento_Precio CHECK (PrecioUnitario >= 0),
    CONSTRAINT CK_Medicamento_Stock CHECK (Stock >= 0)
);

Tabla: Receta
CREATE TABLE Receta (
    RecetaID INT PRIMARY KEY IDENTITY(1,1),
    ConsultaID INT NOT NULL,
    FechaEmision DATETIME NOT NULL DEFAULT GETDATE(),
    Observaciones TEXT,
    Estado VARCHAR(20) DEFAULT 'EMITIDA', --> EMITIDA, DISPENSADA, ANULADA
    CONSTRAINT FK_Receta_Consulta FOREIGN KEY (ConsultaID) 
        REFERENCES Consulta(ConsultaID) ON DELETE CASCADE,
    CONSTRAINT CK_Receta_Estado CHECK (Estado IN ('EMITIDA', 'DISPENSADA', 'ANULADA'))
);

 Tabla intermedia: DetalleReceta
CREATE TABLE DetalleReceta (
    RecetaID INT NOT NULL,
    MedicamentoID INT NOT NULL,
    Dosis VARCHAR(50) NOT NULL,
    Duracion VARCHAR(50) NOT NULL,
    Cantidad INT NOT NULL DEFAULT 1,
    Instrucciones VARCHAR(255),
    CONSTRAINT PK_DetalleReceta PRIMARY KEY (RecetaID, MedicamentoID),
    CONSTRAINT FK_DetalleReceta_Receta FOREIGN KEY (RecetaID) 
        REFERENCES Receta(RecetaID) ON DELETE CASCADE,
    CONSTRAINT FK_DetalleReceta_Medicamento FOREIGN KEY (MedicamentoID) 
        REFERENCES Medicamento(MedicamentoID),
    CONSTRAINT CK_DetalleReceta_Cantidad CHECK (Cantidad > 0)
);

-- 8. MÓDULO DE FACTURACIÓN

 Tabla: Factura
CREATE TABLE Factura (
    FacturaID INT PRIMARY KEY IDENTITY(1,1),
    ConsultaID INT NOT NULL,
    PacienteID INT NOT NULL,
    FechaEmision DATETIME NOT NULL DEFAULT GETDATE(),
    Subtotal DECIMAL(10,2) NOT NULL DEFAULT 0,
    IVA DECIMAL(10,2) NOT NULL DEFAULT 0,
    Total DECIMAL(10,2) NOT NULL DEFAULT 0,
    MetodoPago VARCHAR(50) NOT NULL,
    EstadoPago VARCHAR(20) DEFAULT 'PENDIENTE', --> PENDIENTE, PAGADA, ANULADA
    NumeroFactura VARCHAR(50) UNIQUE,
    CONSTRAINT FK_Factura_Consulta FOREIGN KEY (ConsultaID) 
        REFERENCES Consulta(ConsultaID),
    CONSTRAINT FK_Factura_Paciente FOREIGN KEY (PacienteID) 
        REFERENCES Paciente(PacienteID),
    CONSTRAINT CK_Factura_Total CHECK (Total >= 0),
    CONSTRAINT CK_Factura_MetodoPago CHECK (MetodoPago IN ('EFECTIVO', 'TARJETA', 'TRANSFERENCIA', 'SEGURO')),
    CONSTRAINT CK_Factura_EstadoPago CHECK (EstadoPago IN ('PENDIENTE', 'PAGADA', 'ANULADA'))
);

 Indices para optimización

CREATE INDEX IX_Factura_Fecha ON Factura(FechaEmision);
CREATE INDEX IX_Factura_Paciente ON Factura(PacienteID);

-- 9. MÓDULO DE SEGURIDAD Y USUARIOS

 Tabla: Usuario
CREATE TABLE Usuario (
    UsuarioID INT PRIMARY KEY IDENTITY(1,1),
    NombreUsuario VARCHAR(50) NOT NULL UNIQUE,
    Contrasenia VARCHAR(50) NOT NULL, --> Hash de la contraseña
    Rol VARCHAR(50) NOT NULL,
    Estado VARCHAR(20) DEFAULT 'ACTIVO',
    FechaCreacion DATETIME DEFAULT GETDATE(),
    UltimoAcceso DATETIME,
    ReferenciaID INT, --> FK dinámica según el rol
    TipoReferencia VARCHAR(20), --> 'MEDICO', 'EMPLEADO', 'PACIENTE'
    IntentosFallidos INT DEFAULT 0,
    CONSTRAINT CK_Usuario_Rol CHECK (Rol IN ('ADMINISTRADOR', 'MEDICO', 'EMPLEADO', 'PACIENTE')),
    CONSTRAINT CK_Usuario_Estado CHECK (Estado IN ('ACTIVO', 'INACTIVO', 'BLOQUEADO')),
    CONSTRAINT CK_Usuario_TipoRef CHECK (TipoReferencia IN ('MEDICO', 'EMPLEADO', 'PACIENTE', NULL))
);

 Tabla: AuditoriaAcceso
CREATE TABLE AuditoriaAcceso (
    AuditoriaID INT PRIMARY KEY IDENTITY(1,1),
    UsuarioID INT NOT NULL,
    FechaHora DATETIME DEFAULT GETDATE(),
    TipoAccion VARCHAR(50) NOT NULL,
    IP VARCHAR(45),
    Exitoso BIT NOT NULL,
    Descripcion VARCHAR(255),
    CONSTRAINT FK_Auditoria_Usuario FOREIGN KEY (UsuarioID) 
        REFERENCES Usuario(UsuarioID),
    CONSTRAINT CK_Auditoria_TipoAccion CHECK (TipoAccion IN ('LOGIN', 'LOGOUT', 'INTENTO_FALLIDO', 'CAMBIO_CONTRASENA'))
);

 Índices para optimización
CREATE INDEX IX_Auditoria_Usuario ON AuditoriaAcceso(UsuarioID);
CREATE INDEX IX_Auditoria_Fecha ON AuditoriaAcceso(FechaHora);

###Cambios y Mejoras Implementadas:

Campos adicionales de auditoría:

- FechaCreacion en múltiples tablas
- Estado (BIT) para activación/desactivación lógica
- EstadoConsulta, EstadoPago para mejor control de procesos


Validaciones con CONSTRAINTS:

- Validación de longitud de cédula (10 dígitos)
- Validación de estados permitidos
- Validación de fechas lógicas
- Precios y cantidades no negativas


Mejoras en Facturación:

- Se agrego campos Subtotal e IVA
- NumeroFactura único para control fiscal
- Más opciones de MetodoPago


Mejoras en Medicamentos:

- Campo PrecioUnitario y Stock para inventario
- Campo Cantidad en DetalleReceta


Seguridad mejorada:

- Campo IntentosFallidos en Usuario
- Más tipos de acciones en auditoría
- Campo TipoReferencia para FK dinámica


Índices optimizados para consultas frecuentes
